using System.Reflection;
using Page_switching;

internal static class Program
{
    private static int _checks;
    private static readonly BindingFlags Private = BindingFlags.Instance | BindingFlags.NonPublic;

    [STAThread]
    private static void Main()
    {
        Application.SetHighDpiMode(HighDpiMode.PerMonitorV2);
        Application.EnableVisualStyles();
        var service = new AxisService();
        using var page = new Auto(service);
        Field<System.Windows.Forms.Timer>(page, "_waveUiTimer").Stop();
        var start = Field<AntdUI.Button>(page, "startWaveButton");
        var send = Field<AntdUI.Button>(page, "sendWaveButton");
        var stop = Field<AntdUI.Button>(page, "stopWaveButton");
        var detail = Field<Label>(page, "waveDetailLabel");
        var stage = Field<Label>(page, "waveStageLabel");

        PlcMachineMonitor good = new()
        {
            Version = PlcMachineMonitor.LayoutVersion, Size = PlcMachineMonitor.ByteSize,
            BehaviorVersion = PlcMachineMonitor.ExpectedBehaviorVersion,
            ControlOwner = 2, ControlMode = 1, UploadStation = 2,
            DataReady = true, MotionPermit = true, ConfigValid = true, AxesIdle = true,
            PointCount = 100, ReceivedCount = 100, LoadedCount = 100, ReceivedSequence = 7,
            Heartbeat = 123
        };
        void Reset()
        {
            service.IsConnected = true;
            service.HasControlAuthority = true;
            service.WaveOperationActive = false;
            service.WaveStatus = new("已发送，等待开始", 1, false, true);
            service.LastWaveFailure = null;
            Set(page, "_waveFrame", good);
            Set(page, "_waveImporting", false);
            Set(page, "_waveStopping", false);
            Set(page, "_waveSignal", new WaveTrajectoryFile("offline.csv", new double[] { 0, 0, 0, 0 }));
        }
        void Refresh() => page.GetType().GetMethod("UpdateWaveControls", Private)!.Invoke(page, null);
        void Blocked(Action change, string reason, bool stopAvailable = true)
        {
            Reset();
            change();
            Refresh();
            Check(!start.Enabled, "缺少条件时开始按钮必须禁用：" + reason);
            Check(detail.Text.Contains("暂不能开始") && detail.Text.Contains(reason),
                "实际详情应显示对应的开始阻断原因：" + reason + "；实际：" + detail.Text);
            Check(stop.Enabled == stopAvailable, "停止入口不能跟随开始条件一起禁用：" + reason);
            Check(stage.Text == "已发送，等待开始" || Field<bool>(page, "_waveImporting"),
                "开始原因不应覆盖已有阶段反馈。");
        }

        Reset();
        Refresh();
        Check(start.Enabled && stop.Enabled, "原有开始门槛全部满足时应可开始和停止。");
        Check(!detail.Text.Contains("暂不能开始"), "满足条件时不应遗留开始阻断原因。");
        Check(!good.Axis1.Enabled && !good.Axis2.Enabled && !good.Axis3.Enabled && !good.Axis4.Enabled,
            "场景必须包含四轴尚未使能。");
        Check(!good.Axis1.Homed && !good.Axis2.Homed && !good.Axis3.Homed && !good.Axis4.Homed,
            "场景必须包含四轴尚未回原。");
        Check(start.Enabled, "使能和必要回原由开始流程完成，不能提前用它们禁用开始按钮。");
        Check(good.ControlMode == 1 && start.Enabled, "开始前显示手动模式不能形成自动模式循环门槛。");
        var statistics = Field<AxisMonitoringSummaryControl>(page, "runtimeMonitoringSummary");
        var unconfigured = new AxisMonitoringCounter(7, 2, 123, DateTimeOffset.UtcNow, null, "监测阈值未配置，未统计");
        service.RuntimeMonitoringReport = unconfigured.Snapshot(); Refresh();
        Check(start.Enabled && send.Enabled && stop.Enabled, "监测未配置不参与运动按钮条件。");
        Check(statistics.BuildReportText().Contains("未统计") && statistics.BuildReportText().Contains("未取得有效样本"), "未配置不能显示0%合格。");
        var monitor = new AxisMonitoringCounter(7, 2, 123, DateTimeOffset.UtcNow, new(1, "test", DateTimeOffset.UtcNow,
            Enumerable.Range(0, 4).Select(_ => new AxisMonitoringThreshold(-10, 10, 5)).ToArray()));
        monitor.Fail("模拟统计读取失败"); monitor.Finish(DateTimeOffset.UtcNow, "统计任务结束", true);
        service.RuntimeMonitoringReport = monitor.Snapshot(); Refresh();
        Check(start.Enabled && stop.Enabled && stage.Text == "已发送，等待开始", "统计失败不覆盖运动阶段、不禁用开始停止。");
        Check(statistics.BuildReportText().Contains("统计不完整"), "报告说明统计缺失。");
        service.RuntimeMonitoringReport = null;

        // 每次只打破一个原有门槛，直接观察真实控件 Enabled 和详情 Text。
        Blocked(() => service.IsConnected = false, "控制权", stopAvailable: false);
        Blocked(() => service.HasControlAuthority = false, "控制权", stopAvailable: false);
        Blocked(() => service.WaveOperationActive = true, "正在执行");
        Blocked(() => Set(page, "_waveStopping", true), "停止", stopAvailable: false);
        Blocked(() => Set(page, "_waveImporting", true), "导入");
        Blocked(() => service.WaveStatus = new("已发送，等待开始", 1, false, false), "发送");
        Blocked(() => Set(page, "_waveFrame", null), "反馈");
        var missing = good; missing.DataReady = false;
        Blocked(() => Set(page, "_waveFrame", missing), "拼接");
        missing = good; missing.BehaviorVersion = 0;
        Blocked(() => Set(page, "_waveFrame", missing), "接口版本");
        missing = good; missing.MotionPermit = false;
        Blocked(() => Set(page, "_waveFrame", missing), "运动许可");
        missing = good; missing.ConfigValid = false;
        Blocked(() => Set(page, "_waveFrame", missing), "参数");
        missing = good; missing.AxisFault = true;
        Blocked(() => Set(page, "_waveFrame", missing), "故障");
        missing = good; missing.AxesIdle = false;
        Blocked(() => Set(page, "_waveFrame", missing), "空闲");
        missing = good; missing.UploadStation = 1;
        Blocked(() => Set(page, "_waveFrame", missing), "本站");

        // 页面应提前解释真实执行会拒绝的帧，不能只依靠组汇总标志。
        missing = good; missing.Axis3.ActualPosition = double.NaN;
        Blocked(() => Set(page, "_waveFrame", missing), "轴 3 位置反馈无效");
        missing = good; missing.Axis2.Speed = double.PositiveInfinity;
        Blocked(() => Set(page, "_waveFrame", missing), "轴 2 速度反馈无效");
        missing = good; missing.Axis4.Alarm = true;
        Blocked(() => Set(page, "_waveFrame", missing), "轴 4 报警");
        missing = good; missing.ReceivedSequence = 8;
        Blocked(() => Set(page, "_waveFrame", missing), "接收序号");
        missing = good; missing.AcceptedSequence = 7;
        Blocked(() => Set(page, "_waveFrame", missing), "已启动过");
        missing = good; missing.UploadBusy = true;
        Blocked(() => Set(page, "_waveFrame", missing), "正在拼接");
        missing = good; missing.TrajectoryBusy = true;
        Blocked(() => Set(page, "_waveFrame", missing), "尚未退出");
        missing = good; missing.Error = true; missing.TrajectoryErrorId = 1010;
        Blocked(() => Set(page, "_waveFrame", missing), "1010");
        missing = good; missing.Version = 0;
        Blocked(() => Set(page, "_waveFrame", missing), "布局不匹配");

        // 下发和开始仍是两个步骤；未使能、未回原、未给运动许可不妨碍点位预检与下发。
        Reset();
        missing = good; missing.MotionPermit = false; missing.ConfigValid = false;
        missing.Axis4.Alarm = true; missing.Error = true; missing.TrajectoryErrorId = 1010;
        Set(page, "_waveFrame", missing);
        Refresh();
        Check(send.Enabled && !start.Enabled, "运动准备条件不能误用于禁用发送，真实参数仍在服务预检。");
        Set(page, "_waveSignal", null);
        Refresh();
        Check(!send.Enabled && detail.Text.Contains("暂不能发送：请先导入"), "未导入文件时发送按钮与原因一致。");
        Reset();
        Set(page, "_waveImporting", true);
        Refresh();
        Check(!send.Enabled && detail.Text.Contains("暂不能发送：正在导入"), "导入期间解释发送禁用原因。");

        // 同时欠缺的条件须一起显示，不能只保留第一个，或被“暂不能发送”覆盖。
        Reset();
        service.IsConnected = false;
        service.HasControlAuthority = false;
        service.WaveOperationActive = true;
        service.WaveStatus = new("已发送，等待开始", 1, false, false);
        Set(page, "_waveImporting", true);
        missing = good;
        missing.DataReady = missing.MotionPermit = missing.ConfigValid = missing.AxesIdle = false;
        missing.AxisFault = true;
        missing.UploadStation = 1;
        Set(page, "_waveFrame", missing);
        Refresh();
        Check(!start.Enabled && !stop.Enabled, "多个条件不满足且无有效控制权时，开始和停止均应禁用。");
        foreach (string reason in new[] { "控制权", "正在执行", "导入", "发送", "拼接", "运动许可", "参数", "故障", "空闲", "本站" })
            Check(detail.Text.Contains(reason), "多个阻断原因必须一起展示：" + reason + "；实际：" + detail.Text);
        Check(detail.Text.Contains("PLC 接收") && detail.Text.Contains("PLC 拼接"),
            "补充开始原因时仍须保留PLC接收和拼接进度。");
        int detailChanges = 0;
        detail.TextChanged += (_, _) => detailChanges++;
        Refresh();
        Check(detailChanges == 0, "条件未变化时周期刷新不能反复清除、重写原因导致闪动。");

        Reset();
        Refresh();
        Check(start.Enabled && !detail.Text.Contains("暂不能开始"), "条件恢复后必须移除旧原因并恢复开始按钮。");
        missing = good;
        missing.Error = true;
        missing.TrajectoryErrorId = 1003;
        missing.TrajectoryState = 90;
        Set(page, "_waveFrame", missing);
        service.WaveStatus = new("造波操作失败", 0, false, false, "轨迹启动确认失败：PLC 轨迹故障 1003");
        Refresh();
        foreach (var text in new[] { "1003", "首尾静止", "加速度", "停车空间", "故障等待复位", "最近操作失败", "轨迹启动确认" })
            Check(detail.Text.Contains(text), "实际页面显示故障含义和原始失败阶段：" + text);
        Reset();
        Refresh();
        Check(!detail.Text.Contains("最近操作失败") && !detail.Text.Contains("1003"), "新操作状态清除后不遗留旧故障详情。");
        service.LastWaveFailure = "首点定位失败：轴 2，原始错误 16384";
        service.WaveStatus = new("已停止，再次运行请重新发送", 0, false, false);
        Refresh();
        Check(detail.Text.Contains("历史") && detail.Text.Contains("轴 2") && detail.Text.Contains("16384"),
            "成功停止后仍展示历史原始失败，不能误认为当前仍在报同一故障。");
        Check(stage.Text.StartsWith("已停止") && stop.Enabled, "历史记录不能覆盖当前停车阶段或禁用停止。");
        service.WaveStatus = new("已发送，等待开始", 1, false, true);
        Refresh();
        Check(start.Enabled && detail.Text.Contains("16384"), "历史失败仅供追溯，不作为新的启动门槛。");
        service.WaveStatus = new("造波操作失败", 0, false, false, service.LastWaveFailure);
        Refresh();
        Check(detail.Text.IndexOf("16384") == detail.Text.LastIndexOf("16384"), "同一失败不在当前和历史区重复展示。");
        Check(service.CommandCalls == 0, "更新按钮和原因不能调用启动、停止或上传命令。");
        Console.WriteLine($"Wave start UI checks: {_checks} assertions passed; actual auto.Wave.cs, no PLC access.");
    }

    private static T Field<T>(object target, string name) =>
        (T)(target.GetType().GetField(name, Private)?.GetValue(target) ?? throw new Exception("找不到字段：" + name));
    private static void Set(object target, string name, object? value) =>
        (target.GetType().GetField(name, Private) ?? throw new Exception("找不到字段：" + name)).SetValue(target, value);
    private static void Check(bool condition, string message)
    {
        if (!condition) throw new Exception(message);
        _checks++;
    }
}

namespace Page_switching
{
    // 补齐真实分部文件需要的最少控件，不创建完整页面，也不加载设备服务。
    public partial class Auto : UserControl
    {
        private readonly AntdUI.Button importWaveButton = new();
        private readonly AntdUI.Button sendWaveButton = new();
        private readonly AntdUI.Button startWaveButton = new();
        private readonly AntdUI.Button stopWaveButton = new();
        private readonly Label waveFileLabel = new();
        private readonly Label waveStageLabel = new();
        private readonly Label waveDetailLabel = new();
        private readonly AxisMonitoringSummaryControl runtimeMonitoringSummary = new();
        private readonly ProgressBar waveProgressBar = new() { Maximum = 1000 };
        public Auto()
        {
            _waveFrame = null;
            Controls.AddRange(new Control[] { importWaveButton, sendWaveButton, startWaveButton,
                stopWaveButton, waveFileLabel, waveStageLabel, waveDetailLabel, waveProgressBar });
        }
        public void AddLog(string message) { }
    }

    public sealed record WaveOperationStatus(string Stage, double Fraction, bool Active, bool Ready, string? Error = null);

    // 服务替身没有网络依赖；若页面刷新误发命令，计数会使检查失败。
    public sealed class AxisService
    {
        public bool IsConnected { get; set; } = true;
        public bool HasControlAuthority { get; set; } = true;
        public ushort ControlStationCode => 2;
        public bool WaveOperationActive { get; set; }
        public WaveOperationStatus WaveStatus { get; set; } = new("未导入", 0, false, false);
        public string? LastWaveFailure { get; set; }
        internal AxisMonitoringReport? RuntimeMonitoringReport { get; set; }
        public int CommandCalls { get; private set; }
        internal List<string> GetWaveStartBlockReasons(PlcMachineMonitor frame) =>
            WaveCommandChecks.ForStart(frame, ControlStationCode, WaveStatus.Ready ? 7u : 0u);
        public void ForgetWaveSignal() => CommandCalls++;
        public Task StartWaveAsync() { CommandCalls++; return Task.CompletedTask; }
        public Task StopWaveAsync() { CommandCalls++; return Task.CompletedTask; }
    }

    internal static class OperationJournal { internal static void Record(string page, string message) { } }
}
