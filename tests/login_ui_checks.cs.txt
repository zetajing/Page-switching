using System.ComponentModel;
using System.ComponentModel.Design;
using System.Reflection;
using System.Runtime.Loader;
using System.Runtime.InteropServices;
using System.Windows.Forms;

internal static class Program
{
    static Assembly app = null!;
    static int checks;
    const BindingFlags Members = BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic;

    static void Check(bool passed, string message)
    {
        checks++;
        if (!passed) throw new Exception(message);
    }

    static object Create(string name, params object?[] arguments) =>
        Activator.CreateInstance(app.GetType("Page_switching." + name)!, Members, null, arguments, null)!;
    static object? Call(object target, string name, params object?[] arguments) =>
        target.GetType().GetMethod(name, Members)!.Invoke(target, arguments);
    static Control Field(object target, string name) =>
        (Control)target.GetType().GetField(name, Members)!.GetValue(target)!;
    static object Value(object target, string name) =>
        target.GetType().GetField(name, Members)!.GetValue(target)!;
    static object? Property(object target, string name) => target.GetType().GetProperty(name, Members)!.GetValue(target);

    static void Complete(Task task)
    {
        var limit = DateTime.UtcNow.AddSeconds(20);
        while (!task.IsCompleted)
        {
            if (DateTime.UtcNow > limit) throw new TimeoutException("UI 提交未退出");
            Application.DoEvents();
            Thread.Sleep(5);
        }
        task.GetAwaiter().GetResult();
    }

    static void Render(Control control, string output)
    {
        if (control is not Form)
        {
            using var host = new Form { ClientSize = control.Size, ShowInTaskbar = false, Opacity = 0 };
            host.Controls.Add(control);
            control.Dock = DockStyle.Fill;
            Render(host, output);
            host.Controls.Remove(control);
            return;
        }
        var form = (Form)control;
        form.ShowInTaskbar = false;
        form.Opacity = 0;
        form.Show();
        Application.DoEvents();
        using var bitmap = new System.Drawing.Bitmap(form.Width, form.Height);
        form.DrawToBitmap(bitmap, new System.Drawing.Rectangle(System.Drawing.Point.Empty, bitmap.Size));
        bitmap.Save(output);
        form.Hide();
    }

    static void CheckLayout(Control form, params string[] names)
    {
        foreach (string name in names)
        {
            Control control = Field(form, name);
            Check(control.Parent!.ClientRectangle.Contains(control.Bounds), name + " 越出父容器");
            Check(control.Width > 0 && control.Height >= 25, name + " 尺寸无效");
        }
    }

    // 用真实事件与临时日志核对拆分后的页面归属和过滤规则，不启动通信。
    static void CheckPageLogging()
    {
        const BindingFlags StaticMembers = BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic;
        var journal = app.GetType("Page_switching.OperationJournal")!;
        string logDirectory = (string)journal.GetProperty("DirectoryPath", StaticMembers)!.GetValue(null)!;
        Check(logDirectory.StartsWith(Path.GetTempPath(), StringComparison.OrdinalIgnoreCase), "日志检查使用临时配置目录");
        var journalEvent = journal.GetEvent("EntryAdded", StaticMembers)!;
        var entries = new List<string>();
        Action<string> capture = entry => entries.Add(entry);
        try
        {
            journalEvent.GetAddMethod(true)!.Invoke(null, new object[] { capture });
            using var page = (UserControl)Create("panel.Data");
            var container = new Panel();
            page.Controls.Add(container);
            var controls = Assembly.Load("AntdUI");
            Control Add(string type, string name, string text)
            {
                var control = (Control)Activator.CreateInstance(controls.GetType("AntdUI." + type)!)!;
                control.Name = name;
                control.Text = text;
                container.Controls.Add(control);
                return control;
            }
            var button = Add("Button", "testButton", "调试\r\n操作");
            var jogButton = Add("Button", "jogPositiveButton", "点动");
            var checkBox = Add("Checkbox", "testCheckBox", "测试开关");
            var autoScale = Add("Checkbox", "autoScaleCheckBox", "自动跟随");
            var result = Add("Label", "status", "初始状态");
            var untrackedLabel = Add("Label", "titleLabel", "标题");
            app.GetType("Page_switching.Mainpage")!.GetMethod("TrackActions", StaticMembers)!
                .Invoke(null, new object[] { page, container });

            Call(button, "OnClick", EventArgs.Empty);
            Check(entries.Count == 1 && entries[0].EndsWith("[数据管理] 点击调试  操作"), "按钮日志只记录一次并保留原有换行处理");
            Call(jogButton, "OnClick", EventArgs.Empty);
            Check(entries.Count == 1, "点动按钮仍由原操作流程记录，不重复注册");
            checkBox.GetType().GetProperty("Checked")!.SetValue(checkBox, true);
            Check(entries.Count == 2 && entries[^1].EndsWith("[数据管理] 测试开关：选中"), "复选框变更进入具名日志方法");
            autoScale.GetType().GetProperty("Checked")!.SetValue(autoScale, true);
            Check(entries.Count == 2, "自动跟随仍不记录操作日志");
            page.Visible = false;
            result.Text = "隐藏页结果";
            Check(entries.Count == 3 && entries[^1].EndsWith("[数据管理] 隐藏页结果"), "隐藏页结果保留原页面归属");
            untrackedLabel.Text = "新标题";
            Check(entries.Count == 3, "普通标题变化不产生结果日志");
            string saved = string.Join("\n", Directory.GetFiles(logDirectory, "*.log").Select(File.ReadAllText));
            Check(saved.Contains("[数据管理] 隐藏页结果") && !saved.Contains("写入日志失败"), "日志实际写入临时目录");
        }
        finally
        {
            journalEvent.GetRemoveMethod(true)!.Invoke(null, new object[] { capture });
        }
    }

    [STAThread]
    static void Main(string[] args)
    {
        Application.SetHighDpiMode(HighDpiMode.DpiUnaware);
        Application.EnableVisualStyles();
        // 主窗体仅创建/绘制；设计上下文禁止启动 Router、ADS 和周期刷新。
        LicenseManager.CurrentContext = new DesigntimeLicenseContext();
        string dll = Path.GetFullPath(args[0]);
        // 检查程序从临时目录启动，显式加载应用随附的图形本机库。
        string nativeDirectory = Path.Combine(Path.GetDirectoryName(dll)!, "runtimes",
            "win-" + RuntimeInformation.ProcessArchitecture.ToString().ToLowerInvariant(), "native");
        foreach (string native in Directory.GetFiles(nativeDirectory, "*.dll")) NativeLibrary.Load(native);
        string output = Path.GetFullPath(args[1]);
        Directory.CreateDirectory(output);
        AssemblyLoadContext.Default.Resolving += (_, name) =>
        {
            string path = Path.Combine(Path.GetDirectoryName(dll)!, name.Name + ".dll");
            return File.Exists(path) ? AssemblyLoadContext.Default.LoadFromAssemblyPath(path) : null;
        };
        app = Assembly.LoadFrom(dll);
        string temporary = Path.Combine(Path.GetTempPath(), "PageSwitching_LoginUI_" + Guid.NewGuid().ToString("N"));
        Directory.CreateDirectory(temporary);
        try
        {
            object service = Create("LoginService", Path.Combine(temporary, "users.json"));
            using (var setup = (Form)Create("LoginForm", service))
            {
                Render(setup, Path.Combine(output, "login-first-admin.png"));
                Check(Field(setup, "titleLabel").Text == "创建管理员", "首次创建管理员");
                CheckLayout(setup, "userNameInput", "passwordInput", "confirmPasswordInput", "submitButton", "exitButton");
                Field(setup, "userNameInput").Text = "管理员";
                Field(setup, "passwordInput").Text = "LocalTest-Password01";
                Field(setup, "confirmPasswordInput").Text = "different";
                Complete((Task)Call(setup, "SubmitAsync")!);
                Check(setup.DialogResult != DialogResult.OK && !(bool)Property(service, "HasAccount")!, "两次密码不同不能创建账号");
                Field(setup, "confirmPasswordInput").Text = "LocalTest-Password01";
                Complete((Task)Call(setup, "SubmitAsync")!);
                Check(setup.DialogResult == DialogResult.OK, "创建完成进入系统");
            }
            object admin = Call(service, "ValidateLogin", "管理员", "LocalTest-Password01")!;
            object operatorRole = Enum.Parse(app.GetType("Page_switching.UserRole")!, "Operator");
            Call(service, "AddUser", admin, "操作员", "Operator-Test01", operatorRole);
            object operatorUser = Call(service, "ValidateLogin", "操作员", "Operator-Test01")!;
            using (var login = (Form)Create("LoginForm", service))
            {
                Render(login, Path.Combine(output, "login.png"));
                Check(Field(login, "titleLabel").Text == "用户登录", "后续进入登录模式");
                Field(login, "userNameInput").Text = "操作员";
                Field(login, "passwordInput").Text = "Wrong-Password";
                Complete((Task)Call(login, "SubmitAsync")!);
                Check(login.DialogResult != DialogResult.OK && Field(login, "messageLabel").Text.Contains("不正确"), "错误密码留在登录页");
                Field(login, "passwordInput").Text = "Operator-Test01";
                Complete((Task)Call(login, "SubmitAsync")!);
                Check(login.DialogResult == DialogResult.OK && !(bool)Property(Property(login, "User")!, "IsAdministrator")!, "操作员按真实角色登录");
            }
            using (var manager = (UserControl)Create("panel.AccountManagementPage", service, admin))
            {
                // 单独使用临时账号库验证运行时显示事件；主窗体仍在设计上下文运行。
                var designContext = LicenseManager.CurrentContext;
                LicenseManager.CurrentContext = new LicenseContext();
                try { Render(manager, Path.Combine(output, "account-management.png")); }
                finally { LicenseManager.CurrentContext = designContext; }
                Check(((DataGridView)Field(manager, "accountGrid")).Rows.Count == 2, "账号列表显示两种角色");
                CheckLayout(manager, "accountGrid", "userNameInput", "passwordInput", "confirmPasswordInput", "addButton", "resetButton", "deleteButton");
            }
            foreach (var entry in new[] { (User: admin, Admin: true), (User: operatorUser, Admin: false) })
            {
                using var main = (Form)Create("Mainpage", entry.User);
                // 检查真实对象引用，防止目录拆分后为每个页面或每根轴重复创建控制服务。
                object sharedAxisService = Value(main, "_axisService");
                foreach (var binding in new[] { ("_autoPage", "_waveService"), ("_manualPage", "_axisService"),
                             ("_controlAuthorityPage", "_axisService"), ("_diagnosticsPage", "_axisService") })
                    Check(ReferenceEquals(sharedAxisService, Value(Value(main, binding.Item1), binding.Item2)),
                        "页面共用主窗体轴服务：" + binding.Item1);
                Check(!(bool)Value(Value(main, "_manualPage"), "_ownsAxisService"), "手动页不释放主窗体共享服务");
                foreach (object axis in (System.Collections.IEnumerable)Property(sharedAxisService, "Axes")!)
                {
                    Check(ReferenceEquals(sharedAxisService, Value(axis, "_axisService")), "四轴对象共用同一轴服务");
                    Check(ReferenceEquals(axis, Call(sharedAxisService, "GetAxis", Property(axis, "AxisNumber"))),
                        "GetAxis 返回缓存轴对象");
                }
                Render(main, Path.Combine(output, entry.Admin ? "main-admin.png" : "main-operator.png"));
                Check(Field(main, "manageAccountsButton").Enabled == entry.Admin, "账号管理按钮权限");
                Check(Field(main, "bu_Configuration").Enabled == entry.Admin, "配置按钮权限");
                Check(Field(main, "Bu_Calibration").Enabled == entry.Admin, "标定按钮权限");
                foreach (string page in new[] { "_configPage", "_calibrationPage", "_batchCalibrationPage", "_accountManagementPage", "_monitoringSettingsPage" })
                    Check((bool)Call(main, "CanOpenPage", main.GetType().GetField(page, Members)!.GetValue(main))! == entry.Admin, "统一页面入口权限：" + page);
                foreach (string page in new[] { "_autoPage", "_manualPage", "_dataPage", "_controlAuthorityPage", "_diagnosticsPage" })
                    Check((bool)Call(main, "CanOpenPage", main.GetType().GetField(page, Members)!.GetValue(main))!, "保留操作员功能：" + page);
                // 通过真实入口跳转，检查父容器、返回路径、缓存复用和权限拦截。
                main.Show();
                int windows = Application.OpenForms.Count;
                var container = Field(main, "panelswitch");
                Control Page(string name) => Field(main, name);
                void Current(Control expected, string message)
                {
                    Check(container.Controls.Count == 1 && ReferenceEquals(container.Controls[0], expected), message);
                    Check(expected.Dock == DockStyle.Fill && !expected.IsDisposed, "页面铺满并保留实例");
                    Check(Application.OpenForms.Count == windows, "功能入口不创建独立窗口");
                }
                var authority = Page("_controlAuthorityPage");
                Call(main, "Bu_data_Click", null, EventArgs.Empty);
                Call(Page("_dataPage"), "DatabaseButton_Click", null, EventArgs.Empty);
                var database = Page("_databasePage");
                Current(database, "数据库记录使用主窗体页面，两个角色均可查看");
                var databaseGrid = (DataGridView)Field(database, "grid");
                Check(databaseGrid.ReadOnly && !databaseGrid.AllowUserToAddRows && !databaseGrid.AllowUserToDeleteRows, "数据库查询结果只读");
                Check(((ComboBox)Field(database, "kindPicker")).Items.Count == 8, "八种数据库记录入口");
                Call(database, "QueryButton_Click", null, EventArgs.Empty);
                Application.DoEvents();
                Check(Field(database, "resultStatus").Text.Contains("未启用"), "数据库未配置时明确提示，不尝试连接");
                Call(database, "BackButton_Click", null, EventArgs.Empty);
                Current(Page("_dataPage"), "数据库返回数据管理");
                Call(main, "ControlAuthorityButton_Click", null, EventArgs.Empty);
                Check(authority.GetType().GetField("diagnosticsButton", Members) is null, "控制权页移除诊断按钮");
                Check(Field(main, "diagnosticsButton").Enabled, "操作员和管理员均可打开只读诊断");
                Check(Field(main, "diagnosticsButton").Text == "运行诊断", "独立导航显示明确标题");
                Check(Field(main, "diagnosticsButton").Size == Field(main, "controlAuthorityButton").Size, "诊断导航尺寸与其他导航一致");
                Check(Field(main, "diagnosticsButton").Top > Field(main, "systemSectionLabel").Bottom, "诊断位于系统导航分组");
                // 触发真实Click事件，检查Designer绑定，不能只直接调用处理方法。
                Call(Field(main, "diagnosticsButton"), "OnClick", EventArgs.Empty);
                var diagnostics = Page("_diagnosticsPage");
                Current(diagnostics, "诊断使用主窗体容器");
                Check(Field(main, "controlStatusLabel").Text.Contains("运行诊断"), "顶部显示诊断页面名");
                Check(ReferenceEquals(main.GetType().GetField("_activeNavigationButton", Members)!.GetValue(main), Field(main, "diagnosticsButton")), "诊断使用独立导航高亮");
                Check(Field(main, "navigationMarker").Top == Field(main, "diagnosticsButton").Top, "诊断标记跟随独立按钮");
                Call(diagnostics, "BackButton_Click", null, EventArgs.Empty);
                Current(Page("_autoPage"), "诊断返回自动运行页");
                Check(ReferenceEquals(main.GetType().GetField("_activeNavigationButton", Members)!.GetValue(main), Field(main, "Bu_auto")), "返回后恢复自动页高亮");
                Call(main, "DiagnosticsButton_Click", null, EventArgs.Empty);
                Current(diagnostics, "再次进入复用诊断页");
                if (entry.Admin)
                {
                    Call(main, "bu_Configuration_Click", null, EventArgs.Empty);
                    var config = Page("_configPage");
                    Call(config, "MonitoringSettingsButton_Click", null, EventArgs.Empty);
                    var settings = Page("_monitoringSettingsPage");
                    Current(settings, "监测配置使用主窗体容器");
                    Call(settings, "BackButton_Click", null, EventArgs.Empty);
                    Current(config, "监测配置返回系统配置");
                    Call(config, "MonitoringSettingsButton_Click", null, EventArgs.Empty);
                    Current(settings, "再次进入复用配置页");
                    Call(main, "ManageAccountsButton_Click", null, EventArgs.Empty);
                    var accounts = Page("_accountManagementPage");
                    Current(accounts, "账号管理使用主窗体容器");
                    Check(!Field(main, "navigationMarker").Visible, "账号入口使用顶部按钮高亮");
                    Call(main, "ManageAccountsButton_Click", null, EventArgs.Empty);
                    Field(accounts, "passwordInput").Text = "temporary-password";
                    Call(accounts, "CloseButton_Click", null, EventArgs.Empty);
                    Current(settings, "重复点击账号入口仍返回进入前页面");
                    Check(Field(accounts, "passwordInput").Text.Length == 0, "离开账号页清除密码");
                    Check(Field(main, "navigationMarker").Visible, "返回后恢复左侧导航标记");
                }
                else
                {
                    Call(main, "ManageAccountsButton_Click", null, EventArgs.Empty);
                    Current(diagnostics, "操作员账号入口无效");
                    Call(main, "NavigateTo", Page("_accountManagementPage"), Field(main, "manageAccountsButton"));
                    Current(diagnostics, "统一入口拦截操作员账号页");
                    Call(main, "NavigateTo", Page("_monitoringSettingsPage"), Field(main, "bu_Configuration"));
                    Current(diagnostics, "统一入口拦截操作员监测配置页");
                }
                foreach (var size in new[] { new System.Drawing.Size(1200, 800), new System.Drawing.Size(1500, 1000) })
                {
                    main.ClientSize = size;
                    foreach (string page in entry.Admin
                        ? new[] { "_accountManagementPage", "_monitoringSettingsPage", "_diagnosticsPage" }
                        : new[] { "_diagnosticsPage" })
                    {
                        Call(main, "NavigateTo", Page(page), Field(main, page == "_accountManagementPage" ? "manageAccountsButton"
                            : page == "_monitoringSettingsPage" ? "bu_Configuration" : "diagnosticsButton"));
                        Render(main, Path.Combine(output, $"unified-{entry.Admin}-{page}-{size.Width}.png"));
                        main.Show();
                        Current(Page(page), "调整大小后页面仍在统一容器");
                        var navigation = (ScrollableControl)Field(main, "panel2");
                        var diagnosticEntry = Field(main, "diagnosticsButton");
                        navigation.ScrollControlIntoView(diagnosticEntry);
                        Check(navigation.ClientRectangle.Contains(diagnosticEntry.Bounds), "缩小窗口后可以滚动到诊断导航");
                        Check(!diagnosticEntry.Bounds.IntersectsWith(Field(main, "bu_Configuration").Bounds), "诊断导航与系统配置不重叠");
                    }
                }
                main.Hide();
                CheckLayout(main, "currentUserLabel", "manageAccountsButton");
                Check(!Field(main, "accountPanel").Bounds.IntersectsWith(Field(main, "headerStatusPanel").Bounds), "账号区与设备状态区不重叠");
            }
            using (var scaled = (Form)Create("LoginForm", service))
            {
                scaled.Scale(new System.Drawing.SizeF(1.5F, 1.5F));
                Render(scaled, Path.Combine(output, "login-150-percent.png"));
                CheckLayout(scaled, "userNameInput", "passwordInput", "submitButton", "exitButton");
            }
            CheckPageLogging();
            File.WriteAllText(Path.Combine(temporary, "users.json"), "{ damaged");
            using (var damaged = (Form)Create("LoginForm", service))
            {
                Check(!Field(damaged, "submitButton").Enabled && Field(damaged, "messageLabel").Text.Contains("损坏"), "损坏账号库不能重新注册或进入主窗体");
            }
            Console.WriteLine($"Login UI: {checks} assertions passed; offline only, no Router or PLC connection.");
        }
        finally { Directory.Delete(temporary, recursive: true); }
    }
}
