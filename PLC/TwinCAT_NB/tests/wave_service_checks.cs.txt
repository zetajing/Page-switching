using Page_switching;
using InduLink.Protocols.Ads;
using System.Collections.Specialized;

internal static class Program
{
    static int assertions;
    static void Check(bool value, string message) { assertions++; if (!value) throw new Exception(message); }
    static async Task<Exception> Reject(Func<Task> action, string name)
    {
        try { await action(); throw new Exception("未拒绝：" + name); }
        catch (Exception ex) when (ex is InvalidOperationException or InvalidDataException or OperationCanceledException or TimeoutException or AggregateException) { assertions++; return ex; }
    }
    static async Task Until(Func<bool> ready)
    {
        using var limit = new CancellationTokenSource(5000);
        while (!ready()) await Task.Delay(20, limit.Token);
    }
    // 原有运动回归不读取用户监测配置，也不创建真实监测连接。
    static AxisService NewService(AxisServiceOptions options) => new(options, new AxisRuntimeMonitoringService(
        () => throw new Exception("离线回归不允许创建监测来源"), new AxisMonitoringClock(),
        () => (null, "离线监测阈值未配置"), Path.Combine(AppContext.BaseDirectory, "monitoring"), _ => { }));
    static AxisServiceOptions Options(ushort station = 1) => new()
    {
        ControlStationCode = station,
        AxisSymbols = Enumerable.Range(1, 4).Select(i => new AxisSymbolMap
        {
            EnableCommand = $"GVL_control.aStation[1].aAxis[{i}].xStart",
            HomeCommand = $"GVL_control.aStation[1].aAxis[{i}].gohome",
            StopCommand = $"GVL_control.aStation[1].aAxis[{i}].xStop",
            ResetAlarmCommand = $"GVL_control.aStation[1].aAxis[{i}].reset",
            JogPositiveCommand = $"GVL_control.aStation[1].aAxis[{i}].jogz",
            JogNegativeCommand = $"GVL_control.aStation[1].aAxis[{i}].jogf",
            JogSpeedCommand = $"GVL_control.aStation[1].aAxis[{i}].lrJogSpeed"
        }).ToArray()
    };
    static async Task Main()
    {
        var temp = Path.GetTempFileName();
        try
        {
            string FileText(int count = 4, string dt = "0.002", string header = "OldSignal", string? data = null)
                => header + "\n" + count + "\n" + dt + "\n1\n1\n1\n0\n0\n0\n" + (data ?? "1,999\n1,999\n2,999\n2,999\n");
            File.WriteAllText(temp, FileText());
            var signal = WaveTrajectoryFile.Read(temp);
            Check(signal.Positions.SequenceEqual(new[] { 1d, 1, 2, 2 }), "只使用第一列，不缩放");
            Check(signal.DurationSeconds == 0.006, "duration");
            foreach (var text in new[] { FileText(dt: "0.01"), FileText(count: 3), FileText(count: 2100001),
                FileText(header: "TimeSeconds,ElevationMetres"), FileText(data: "1\nNaN\n2\n2\n"),
                FileText(data: "1\nInfinity\n2\n2\n"), FileText(data: ",999\n1\n2\n2\n"), FileText(data: "1\n1\n2\n"), FileText(data: "1\n1\n2\n2\n3\n") })
            {
                File.WriteAllText(temp, text);
                await Reject(() => Task.FromResult(WaveTrajectoryFile.Read(temp)), "invalid file");
            }
            foreach (int count in new[] { 4, 99999, 100000, 100001, 2100000 })
            {
                var chunks = WaveTransferChunks.Create(count).ToArray();
                Check(chunks.Sum(c => (long)c.Count) == count * 4L, "four-axis count");
                Check(chunks.Count(c => c.EndOfSegment) == (count + 99999) / 100000, "one commit per segment");
                foreach (var chunk in chunks)
                {
                    if (chunk.Count <= 0 || chunk.Count > 1024 || chunk.Point + chunk.Count > chunk.SegmentCount
                        || chunk.Offset != chunk.Segment * 100000 + chunk.Point || chunk.Axis is < 1 or > 4)
                        throw new Exception("chunk range");
                }
                assertions++;
            }
            // 只读连接不需要站点身份或控制权；读取不会触发申请、心跳或运动写入。
            AdsClient.Model = new FakePlc { AuthorityOwner = 0 };
            using (var service = NewService(Options(station: 0)))
            {
                var disconnected = await service.ReadSnapshotAsync(CancellationToken.None);
                Check(disconnected.Count == 4 && disconnected.All(s => s.ActualPosition is null), "disconnected read is unavailable");
                await service.ConnectAsync(CancellationToken.None);
                AdsClient.Model.Frame.Axis1.ActualPosition = 12.5;
                var snapshots = await service.ReadSnapshotAsync(CancellationToken.None);
                Check(!service.HasControlAuthority && snapshots[0].ActualPosition == 12.5, "read without authority");
                var diagnostic = await service.ReadDiagnosticsAsync(CancellationToken.None);
                Check(!service.HasControlAuthority && diagnostic.Limits!.Length == 4, "read all parameters without authority");
                Check(diagnostic.Frame.Axis1.ActualPosition == 12.5 && diagnostic.Limits![0].MinPosition == -200,
                    "diagnostic uses real frame and PLC configuration");
                Check(diagnostic.Format(null).Contains("PLC/NC 原始单位"), "diagnostic does not infer mechanical units");
                AdsClient.Model.InvalidConfigRead = true;
                diagnostic = await service.ReadDiagnosticsAsync(CancellationToken.None);
                Check(diagnostic.Limits is null && diagnostic.ParameterError!.Contains("aConfig"), "invalid parameters stay unavailable, frame remains readable");
                AdsClient.Model.InvalidConfigRead = false;
                Check(ReferenceEquals(service.GetAxis(1).Snapshot, snapshots[0]), "fixed axis receives latest read");
                Check(!snapshots[0].IsEnabled, "enable display uses actual feedback");
                await Reject(() => service.EnableAllAsync(CancellationToken.None), "manual command without authority");
                Check(AdsClient.Model.TotalWrites == 0, "read and rejected command never write PLC");
                AdsClient.Model.Frame.Version = 0;
                snapshots = await service.ReadSnapshotAsync(CancellationToken.None);
                Check(snapshots.All(s => s.ActualPosition is null), "invalid frame never retains valid display");
                Check(service.LastError!.Contains("GVL_HMI.stMonitor") && service.LastError.Contains("布局不匹配"), "read failure keeps symbol and original layout reason");
                Check(service.ConnectionStateText.Contains(service.LastError), "connected display includes actual read error");
                int logCount = OperationJournal.Entries.Count;
                await service.ReadSnapshotAsync(CancellationToken.None);
                Check(OperationJournal.Entries.Count == logCount, "same read error does not flood operation log");
                AdsClient.Model.Frame.Version = PlcMachineMonitor.LayoutVersion;
                await service.ReadSnapshotAsync(CancellationToken.None);
                Check(service.LastError is null, "successful read clears old error");
                Check(OperationJournal.Entries.Count == logCount + 1, "read recovery recorded once");
                await service.ReadSnapshotAsync(CancellationToken.None);
                Check(OperationJournal.Entries.Count == logCount + 1, "normal refresh does not repeat recovery log");
                Check(AdsClient.Model.TotalWrites == 0, "read failure never attempts reset or reconnect write");
            }
            AdsClient.Model = new FakePlc();
            using (var service = NewService(Options()))
            {
                await Reject(() => service.UploadWaveAsync(signal), "no authority");
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                // NC 反馈故障不撤销本站复位权限；上位机必须收到独立故障标志。
                AdsClient.Model.FeedbackInvalid = true;
                await Until(() => service.AuthorityStatus.FeedbackInvalid);
                Check(service.HasControlAuthority && !service.AuthorityStatus.Locked, "invalid feedback alone keeps control available");
                await service.ResetAlarmsAsync(CancellationToken.None);
                Check(AdsClient.Model.ResetWrites == 4, "authorized reset reaches all four mailboxes");
                AdsClient.Model.FeedbackInvalid = false;
                await Until(() => !service.AuthorityStatus.FeedbackInvalid);
                await service.UploadWaveAsync(signal);
                var model = AdsClient.Model;
                Check(service.WaveStatus.Ready && model.Frame.DataReady, "upload waits for ready");
                Check(model.ArrayWrites == 16 && model.StartWrites == 0, "upload cannot start");
                Check(model.ValuesByAxis.Values.All(v => v.SequenceEqual(signal.Positions)), "same points to four axes");
                await service.StartWaveAsync();
                Check(model.HomedAxes.SequenceEqual(new[] { 2 }), "only unhomed axis homed");
                Check(model.StartWrites == 1 && !service.WaveStatus.Ready, "single execution consumed");
                Check(model.WaveHeartbeatWrites > 0, "trajectory heartbeat independent of page");
                await Reject(() => service.StartWaveAsync(), "no replay");
                await service.UploadWaveAsync(signal);
                model.HoldHoming = true;
                model.Frame.Axis2.Homed = false;
                var pending = service.StartWaveAsync();
                await Until(() => service.WaveStatus.Stage.Contains("回原"));
                await Reject(() => service.UploadWaveAsync(signal), "duplicate operation");
                await Reject(() => service.JogAsync(1, true, true, 1, CancellationToken.None), "manual during preparation");
                await service.StopWaveAsync();
                try { await pending; } catch (OperationCanceledException) { }
                Check(model.StartWrites == 1, "cancelled home never writes start");
                Check(service.WaveStatus.Stage.StartsWith("已停止"), "stop confirmed after release");
                Check(!model.Session && !model.Frame.TrajectoryBusy, "stop releases session");
                Check(model.StopReleaseWrites > 0, "stop latch released before idle confirmation");
            }
            // 统计来源/日志异常不改变真实启动、运动结果或停止入口；不连接第二个真实 ADS。
            AdsClient.Model = new FakePlc();
            var monitoring = new AxisRuntimeMonitoringService(() => throw new IOException("模拟只读采样连接失败"),
                new AxisMonitoringClock(), () => (new AxisMonitoringProfile(1, "integration", DateTimeOffset.UtcNow,
                    Enumerable.Range(0, 4).Select(_ => new AxisMonitoringThreshold(-10, 10, 5)).ToArray()), null),
                Path.Combine(AppContext.BaseDirectory, "monitoring-error"), _ => throw new IOException("模拟统计日志失败"));
            using (var service = new AxisService(Options(), monitoring))
            {
                await service.ConnectAsync(CancellationToken.None); await Until(() => service.HasControlAuthority);
                await service.UploadWaveAsync(signal); await service.StartWaveAsync();
                await Until(() => service.RuntimeMonitoringReport?.ReadFailures > 0);
                Check(service.WaveStatus.Stage.StartsWith("造波完成") && service.LastWaveFailure is null,
                    "monitoring failure cannot overwrite motion success or fault field");
                Check(AdsClient.Model.StartWrites == 1, "monitoring never adds start, replay or other motion writes");
                await service.StopWaveAsync();
                Check(service.WaveStatus.Stage.StartsWith("已停止") && service.LastWaveFailure is null,
                    "monitoring failure cannot block stop or replace stop confirmation");
                await service.ShutdownRuntimeMonitoringAsync();
            }
            // 页面与真实服务必须解释同一份启动条件；拒绝时不能抢先使能、回原或发启动沿。
            AdsClient.Model = new FakePlc();
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                var model = AdsClient.Model;
                var cases = new (Action<FakePlc> Change, string Reason)[]
                {
                    (p => p.Frame.Axis3.ActualPosition = double.NaN, "轴 3 位置反馈无效"),
                    (p => p.Frame.Axis2.Speed = double.PositiveInfinity, "轴 2 速度反馈无效"),
                    (p => p.Frame.Axis4.Alarm = true, "轴 4 报警"),
                    (p => p.Frame.ReceivedSequence++, "接收序号"),
                    (p => p.Frame.AcceptedSequence = p.Frame.ReceivedSequence, "已启动过"),
                    (p => p.Frame.UploadBusy = true, "正在拼接"),
                    (p => p.Frame.MotionPermit = false, "运动许可"),
                    (p => { p.Frame.Error = true; p.Frame.TrajectoryErrorId = 1010; }, "1010")
                };
                foreach (var item in cases)
                {
                    await service.UploadWaveAsync(signal);
                    var healthy = model.Frame;
                    item.Change(model);
                    var reasons = service.GetWaveStartBlockReasons(model.Frame);
                    Check(reasons.Any(reason => reason.Contains(item.Reason)), "shared start reason: " + item.Reason);
                    int motionWrites = model.MotionWrites, clearWrites = model.ClearWrites;
                    var failure = await Reject(() => service.StartWaveAsync(), item.Reason);
                    Check(failure.Message == string.Join(Environment.NewLine, reasons), "service and page use identical reasons");
                    Check(model.MotionWrites == motionWrites && model.ClearWrites == clearWrites,
                        "rejected preparation cannot issue motion or clear points: " + item.Reason);
                    model.Frame = healthy;
                }
                // 下发阶段不使用运动许可/报警作为门槛；仍读取真实参数并执行数值预检。
                model.Frame.MotionPermit = false;
                model.Frame.Axis4.Alarm = true;
                await service.UploadWaveAsync(signal);
                Check(service.WaveStatus.Ready && model.MotionWrites == 0, "upload remains separate from motion preparation");
            }
            // 单轴与全轴共用脉冲实现：保持轴映射，取消后清零，双重故障保留原始异常。
            AdsClient.Model = new FakePlc();
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                var model = AdsClient.Model;
                await service.ResetAxisAlarmAsync(4, CancellationToken.None);
                Check(model.PulseWrites.TakeLast(2).SequenceEqual(new[] { "GVL_control.aStation[1].aAxis[4].reset=True", "GVL_control.aStation[1].aAxis[4].reset=False" }),
                    "axis four pulse keeps exact mailbox and true/false order");
                await service.ResetAlarmsAsync(CancellationToken.None);
                var expected = Enumerable.Range(1, 4).Select(i => $"GVL_control.aStation[1].aAxis[{i}].reset=True")
                    .Concat(Enumerable.Range(1, 4).Select(i => $"GVL_control.aStation[1].aAxis[{i}].reset=False"));
                Check(model.PulseWrites.TakeLast(8).SequenceEqual(expected), "all-axis pulse preserves write order");
                using var cancel = new CancellationTokenSource();
                model.CancelPulseOnSet = cancel;
                var cancelled = await Reject(() => service.ResetAxisAlarmAsync(3, cancel.Token), "cancel pulse after set");
                Check(cancelled is OperationCanceledException && model.PulseWrites.Last().EndsWith("aAxis[3].reset=False"), "cancellation still clears original axis");
                model.CancelPulseOnSet = null;
                foreach (var mode in new[] { "set", "reset", "both" })
                {
                    model.FailPulseSet = mode != "reset";
                    model.FailPulseReset = mode != "set";
                    var failure = await Reject(() => service.ResetAxisAlarmAsync(2, CancellationToken.None), "pulse failure " + mode);
                    Check(failure.Message.Contains("aAxis[2].reset"), "manual failure contains exact command symbol");
                    Check(model.PulseWrites.Last().EndsWith("aAxis[2].reset=False"), "failed set still attempts reset");
                    if (mode == "both")
                    {
                        Check(failure is AggregateException aggregate && aggregate.InnerExceptions.Count == 2,
                            "set and reset failures remain separate exceptions");
                        Check(failure.ToString().Contains("模拟置位失败") && failure.ToString().Contains("模拟撤销失败"),
                            "cleanup cannot overwrite original set failure");
                    }
                }
                model.FailPulseSet = model.FailPulseReset = false;
                using var cancelWithResetFailure = new CancellationTokenSource();
                model.CancelPulseOnSet = cancelWithResetFailure;
                model.FailPulseReset = true;
                var both = await Reject(() => service.ResetAxisAlarmAsync(4, cancelWithResetFailure.Token), "cancel and reset failure");
                Check(both is AggregateException cancelledAndReset
                    && cancelledAndReset.InnerExceptions[0] is OperationCanceledException,
                    "failed cancellation cleanup cannot be silently treated as normal cancellation");
                Check(both.Message.Contains("撤销手动命令失败") && both.Message.Contains("aAxis[4].reset"), "cancel cleanup identifies uncertain request bit");
                model.CancelPulseOnSet = null;
                model.FailPulseReset = false;
                await service.ResetAxisAlarmAsync(1, CancellationToken.None);
                Check(model.PulseWrites.Last().EndsWith("aAxis[1].reset=False"), "failed pulse always releases service gate");
                var defaults = AxisServiceOptions.FromConfiguration(new NameValueCollection());
                Check(defaults.Unit == "原始单位", "unknown PLC axis unit is not assumed to be degrees");
            }
            // 拆分会话方法后，取消等待 PLC 会话锁仍不能越过握手直接进入清表/下发。
            AdsClient.Model = new FakePlc { HoldSessionLock = true };
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                using var cancelHandshake = new CancellationTokenSource();
                var pending = service.UploadWaveAsync(signal, cancelHandshake.Token);
                await Until(() => AdsClient.Model.Session);
                Check(AdsClient.Model.ArrayWrites == 0 && AdsClient.Model.ClearWrites == 0, "session lock confirmation precedes table changes");
                cancelHandshake.Cancel();
                var failure = await Reject(() => pending, "cancel during session lock handshake");
                Check(failure is OperationCanceledException, "handshake cancellation keeps original cancellation kind");
                Check(!service.WaveOperationActive && !AdsClient.Model.Session, "cancelled handshake completes cleanup before releasing task busy");
                Check(AdsClient.Model.MotionWrites == 0 && AdsClient.Model.ArrayWrites == 0 && AdsClient.Model.ClearWrites == 0,
                    "cancelled handshake cannot produce late motion, clear or point writes");
                AdsClient.Model.HoldSessionLock = false;
                await service.UploadWaveAsync(signal);
                Check(service.WaveStatus.Ready && !AdsClient.Model.Session, "new operation can complete after cancelled handshake");
            }
            AdsClient.Model = new FakePlc { FailArrayWrite = true };
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                var failure = await Reject(() => service.UploadWaveAsync(signal), "write failure");
                Check(failure.Message.Contains("GVL_motion.aUpload") && failure.Message.Contains("轴 1"), "array write failure identifies symbol and axis");
                Check(failure.InnerException?.Message == "模拟 ADS 写入失败", "array write keeps original cause");
                Check(service.WaveStatus.Error == failure.Message && !service.WaveOperationActive, "cleanup retains original failure on page");
                Check(!service.WaveStatus.Ready && AdsClient.Model.StartWrites == 0, "failed transfer not ready");
            }
            AdsClient.Model = new FakePlc();
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                var failure = await Reject(() => service.UploadWaveAsync(new("invalid.csv", new[] { 0d, 0, 1000, 1000 })), "preflight rejects unsafe points");
                Check(failure.Message.Contains("轴") && failure.Message.Contains("点"), "preflight failure identifies axis and point");
                Check(AdsClient.Model.ArrayWrites == 0 && AdsClient.Model.PointMetadataWrites == 0
                    && AdsClient.Model.ClearWrites == 0 && AdsClient.Model.StartWrites == 0, "rejected points never clear old table or write point metadata/data/start");
                Check(!service.WaveStatus.Ready && service.WaveStatus.Error == failure.Message, "preflight failure remains visible");
                Check(!AdsClient.Model.Session, "preflight failure releases session without stop command");
                using var cancelledPreflight = new CancellationTokenSource();
                cancelledPreflight.Cancel();
                await Reject(() => service.UploadWaveAsync(signal, cancelledPreflight.Token), "preflight cancelled");
                Check(!service.WaveStatus.Active && !service.WaveOperationActive
                    && service.WaveStatus.Stage.Contains("已取消"), "preflight cancellation leaves a final inactive state");
                Check(AdsClient.Model.ArrayWrites == 0 && AdsClient.Model.ClearWrites == 0, "cancelled preflight never writes or clears points");
                await service.UploadWaveAsync(signal);
                AdsClient.Model.AxisLimits[2][2] = 999;
                int startWrites = AdsClient.Model.StartWrites;
                failure = await Reject(() => service.StartWaveAsync(), "parameters changed after upload");
                Check(failure.Message.Contains("参数") && failure.Message.Contains("重新发送"), "changed parameters require a new preflight");
                Check(AdsClient.Model.StartWrites == startWrites, "changed parameters cannot send start");
            }
            AdsClient.Model = new FakePlc { InvalidConfigRead = true };
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                var failure = await Reject(() => service.UploadWaveAsync(signal), "config read invalid");
                Check(failure.Message.Contains("GVL_motion.aConfig[1].lrMinPosition"), "config failure identifies exact parameter");
                Check(AdsClient.Model.ArrayWrites == 0 && AdsClient.Model.ClearWrites == 0, "missing config cannot use defaults or send points");
            }
            AdsClient.Model = new FakePlc { ChangeConfigOnHome = true };
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                await service.UploadWaveAsync(signal);
                var failure = await Reject(() => service.StartWaveAsync(), "parameters changed during preparation");
                Check(failure.Message.Contains("参数") && failure.Message.Contains("重新发送"), "parameters are rechecked after homing");
                Check(AdsClient.Model.StartWrites == 0 && AdsClient.Model.StopReleaseWrites > 0, "changed preparation parameters stop safely before start pulse");
            }
            AdsClient.Model = new FakePlc { HoldHoming = true };
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                await service.UploadWaveAsync(signal);
                var pending = service.StartWaveAsync();
                await Until(() => service.WaveStatus.Stage.Contains("回原"));
                AdsClient.Model.Frame.ControlOwner = 2;
                await Reject(() => pending, "owner changed during preparation");
                Check(AdsClient.Model.StartWrites == 0 && !service.HasControlAuthority, "owner loss cannot start");
                AdsClient.Model.Frame.ControlOwner = 1;
                await Task.Delay(300);
                Check(!service.HasControlAuthority, "failed stop cannot auto restore control");
            }
            AdsClient.Model = new FakePlc { HoldAcknowledgement = true };
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                var failure = await Reject(() => service.UploadWaveAsync(signal), "ack timeout");
                Check(failure is TimeoutException && failure.Message.Contains("等待本段接收确认")
                    && failure.Message.Contains("接收=0/4") && failure.Message.Contains("PLC 状态=0"), "timeout includes operation and last valid frame");
                Check(!service.WaveStatus.Ready && AdsClient.Model.StartWrites == 0, "unconfirmed segment never ready");
            }
            AdsClient.Model = new FakePlc { FreezeAfterArray = true, HoldAcknowledgement = true };
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                await Reject(() => service.UploadWaveAsync(signal), "stalled heartbeat");
                Check(!service.HasControlAuthority && AdsClient.Model.StartWrites == 0, "uncertain stop suspends authorization");
            }
            AdsClient.Model = new FakePlc { FeedbackInvalid = true, ReleaseAllowed = false };
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                await Reject(() => service.ChangeControlAuthorityAsync(true, CancellationToken.None), "release permission absent");
                Check(AdsClient.Model.AuthorityRequestWrites == 0, "blocked release never writes request");
                AdsClient.Model.ReleaseAllowed = true;
                await Until(() => service.AuthorityStatus.CanRelease);
                Check(!service.AuthorityStatus.Locked, "one idle lock permits request and release");
                await service.ChangeControlAuthorityAsync(true, CancellationToken.None);
                Check(AdsClient.Model.AuthorityOwner == 0 && !service.HasControlAuthority, "release waits for PLC ack and owner zero");
                Check(service.AuthorityStatus.FeedbackInvalid, "release does not pretend I/O fault cleared");
            }
            // 故障快照必须来自失败当时，不能用安全停止后的状态冒充故障现场。
            AdsClient.Model = new FakePlc { FailWriteSymbol = "GVL_motion.nPointCount" };
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                var failure = await Reject(() => service.UploadWaveAsync(signal), "scalar write failure");
                Check(failure.Message.Contains("GVL_motion.nPointCount") && failure.InnerException?.Message == "模拟标量写入失败", "write diagnostic retains exact symbol and cause");
                Check(AdsClient.Model.StartWrites == 0, "write diagnostic never retries or starts trajectory");
            }
            AdsClient.Model = new FakePlc { FailFrameRead = true };
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                var failure = await Reject(() => service.UploadWaveAsync(signal), "frame read failure");
                Check(failure.Message.Contains("GVL_HMI.stMonitor") && failure.InnerException?.Message == "模拟监控读取失败", "read diagnostic retains exact symbol and cause");
                Check(AdsClient.Model.StartWrites == 0, "read diagnostic never starts trajectory");
            }
            AdsClient.Model = new FakePlc { StartFaultCode = 1003 };
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                await service.UploadWaveAsync(signal);
                var failure = await Reject(() => service.StartWaveAsync(), "trajectory validation fault");
                foreach (var text in new[] { "轨迹启动确认", "1003", "首尾静止", "加速度", "停车空间", "PLC 状态=90", "执行点=0/4" })
                    Check(failure.Message.Contains(text), "fault context includes " + text);
                Check(failure.InnerException is not null, "fault keeps original exception chain");
                Check(!failure.Message.Contains("校验点=0"), "execution point is never presented as validation point");
                Check(service.WaveStatus.Error == failure.Message && service.WaveStatus.Stage == "造波操作失败", "confirmed cleanup cannot erase original fault");
                Check(AdsClient.Model.StartWrites == 1 && !service.WaveStatus.Ready, "fault never retries start or reuses table");
                Check(OperationJournal.Entries.Any(e => e.Contains("StartWaveCoreAsync"))
                    && OperationJournal.Entries.Any(e => e.Contains("1003")), "operation journal retains fault and call stack");
            }
            // 配套错误不能清表、写点位或自动使能；停止仍保留独立入口。
            AdsClient.Model = new FakePlc();
            AdsClient.Model.Frame.BehaviorVersion = 0;
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                var failure = await Reject(() => service.UploadWaveAsync(signal), "incompatible behavior on upload");
                Check(failure.Message.Contains("接口版本不匹配"), "upload reports precise contract mismatch");
                Check(AdsClient.Model.ClearWrites == 0 && AdsClient.Model.ArrayWrites == 0
                    && AdsClient.Model.PointMetadataWrites == 0 && AdsClient.Model.StartWrites == 0, "contract mismatch never mutates point table");
                Check(service.LastWaveFailure == failure.Message, "original failure retained independently of operation status");
            }
            AdsClient.Model = new FakePlc();
            using (var service = NewService(Options()))
            {
                await service.ConnectAsync(CancellationToken.None);
                await Until(() => service.HasControlAuthority);
                await service.UploadWaveAsync(signal);
                AdsClient.Model.Frame.BehaviorVersion = 2;
                await Reject(() => service.StartWaveAsync(), "incompatible behavior before preparation");
                Check(AdsClient.Model.StartWrites == 0 && !AdsClient.Model.Frame.Axis1.Enabled,
                    "changed behavior prevents enable and start");
                await service.StopWaveAsync();
                Check(service.LastWaveFailure!.Contains("接口版本不匹配"), "successful stop does not erase original failure");
            }
            var fault = new PlcMachineMonitor { Error = true, FaultAxis = 2, FaultSource = 10,
                FaultState = 15, TrajectoryState = 60, TrajectoryErrorId = 0x4000, CleanupErrorId = 1009 };
            string faultText = PlcWaveDiagnostics.DescribeFirstFault(fault);
            Check(faultText.Contains("轴 2") && faultText.Contains("MC_MoveAbsolute") && faultText.Contains("定位到首点"), "first fault retains axis, block and original stage");
            Check(faultText.Contains("16384") && faultText.Contains("1009"), "original error and cleanup error both present");
            foreach (uint code in Enumerable.Range(1001, 12).Select(i => (uint)i))
                Check(PlcWaveDiagnostics.DescribeFault(code).Contains(code.ToString()), "known code preserved " + code);
            Check(PlcWaveDiagnostics.DescribeFault(0x4000).Contains("16384")
                && PlcWaveDiagnostics.DescribeFault(0x4000).Contains("0x4000"), "unmapped MC error retains decimal and hex");
            Check(PlcWaveDiagnostics.DescribeState(30) == "执行轨迹", "PLC state is distinct from page state");
            Console.WriteLine($"Wave service: {assertions} assertions passed (fake ADS only).");
        }
        finally { File.Delete(temp); }
    }

namespace InduLink.Protocols.Ads
{
    using InduLink.Abstractions;
    public class AdsClientOptions
    {
        public string DeviceId { get; set; } = "";
        public string AmsNetId { get; set; } = "";
        public int Port { get; set; }
        public int ConnectTimeoutMilliseconds { get; set; }
        public int OperationTimeoutMilliseconds { get; set; }
        public bool ValidateTargetStateOnConnect { get; set; }
        public bool EnableSumCommands { get; set; }
    }
    public sealed class AdsClient(AdsClientOptions options) : IDisposable
    {
        public static FakePlc Model = new();
        public string DeviceId => options.DeviceId;
        public bool IsConnected { get; private set; }
        public Task ConnectAsync(CancellationToken token) { IsConnected = true; return Task.CompletedTask; }
        public Task DisconnectAsync(CancellationToken token) { IsConnected = false; return Task.CompletedTask; }
        public Task WriteAsync(WriteRequest request, CancellationToken token)
        {
            token.ThrowIfCancellationRequested();
            lock (Model) Model.Write(request);
            return Task.CompletedTask;
        }
        public async Task WriteManyAsync(IReadOnlyCollection<WriteRequest> requests, CancellationToken token)
        { foreach (var request in requests) await WriteAsync(request, token); }
        public Task<ReadResult> ReadManyAsync(IReadOnlyCollection<ReadRequest> requests, CancellationToken token)
        {
            if (requests.First().Address.StartsWith("GVL_motion.aConfig["))
            {
                var values = Model.AxisLimits.SelectMany(axis => axis).Select(value => new ReadValue(value)).ToArray();
                if (Model.InvalidConfigRead) values[0] = new ReadValue("invalid");
                return Task.FromResult(new ReadResult(values));
            }
            lock (Model) return Task.FromResult(new ReadResult(new[] { new ReadValue(Model.AuthorityOwner),
                new ReadValue(Model.Session && !Model.HoldSessionLock || !Model.ReleaseAllowed), new ReadValue(Model.AuthorityOwner == 1), new ReadValue(Model.AuthorityAck),
                new ReadValue((ushort)1), new ReadValue(Model.FeedbackInvalid) }));
        }
        public Task<T> ReadAnyAsync<T>(string symbol, int[]? args = null, CancellationToken cancellationToken = default)
        {
            cancellationToken.ThrowIfCancellationRequested();
            if (Model.FailFrameRead) throw new InvalidOperationException("模拟监控读取失败");
            lock (Model) return Task.FromResult((T)(object)Model.Read());
        }
        public void Dispose() { IsConnected = false; }
    }
    public sealed class FakePlc
    {
        public int TotalWrites;
        public bool FeedbackInvalid;
        public int ResetWrites;
        public bool ReleaseAllowed = true;
        public ushort AuthorityOwner = 1;
        public uint AuthorityAck;
        public int AuthorityRequestWrites;
        private ushort authorityAction;
        public PlcMachineMonitor Frame = new()
        {
            Version = PlcMachineMonitor.LayoutVersion, Size = PlcMachineMonitor.ByteSize, ControlOwner = 1,
            BehaviorVersion = PlcMachineMonitor.ExpectedBehaviorVersion, ProgramVersion = 2026102301,
            MotionPermit = true, ConfigValid = true, AxesIdle = true, AxesStopped = true,
            Axis1 = new() { Homed = true }, Axis3 = new() { Homed = true }, Axis4 = new() { Homed = true }
        };
        public bool Session, HoldHoming, FailArrayWrite, HoldAcknowledgement, FrozenHeartbeat, FreezeAfterArray;
        public bool HoldSessionLock;
        public uint StartFaultCode;
        public string? FailWriteSymbol;
        public bool FailFrameRead;
        public bool InvalidConfigRead;
        public bool ChangeConfigOnHome;
        public double[][] AxisLimits = Enumerable.Range(1, 4).Select(_ => new[] { -200d, 200, 1000, 1e6, 1000, 10, 10 }).ToArray();
        public int PointMetadataWrites, ClearWrites;
        public int ArrayWrites, StartWrites, StopReleaseWrites, WaveHeartbeatWrites;
        public int MotionWrites;
        public readonly List<string> PulseWrites = new();
        public bool FailPulseSet, FailPulseReset;
        public CancellationTokenSource? CancelPulseOnSet;
        public Dictionary<int, List<double>> ValuesByAxis = new();
        public List<int> HomedAxes = new();
        private uint sequence, offset, size;
        private int runReads;
        public void Write(WriteRequest request)
        {
            TotalWrites++;
            if (request.Address.EndsWith(".reset") && request.Value is true) ResetWrites++;
            string address = request.Address;
            if (address.Contains(".aAxis[") && address.EndsWith(".reset"))
            {
                PulseWrites.Add(address + "=" + request.Value);
                if (request.Value is true && FailPulseSet) throw new InvalidOperationException("模拟置位失败");
                if (request.Value is false && FailPulseReset) throw new InvalidOperationException("模拟撤销失败");
                if (request.Value is true) CancelPulseOnSet?.Cancel();
            }
            if (request.Value is true && (address == "GVL_motion.xStart" || address == "GVL_motion.xStop"
                || address.Contains(".aAxis[") && (address.EndsWith(".xStart") || address.EndsWith(".gohome") || address.EndsWith(".xStop"))))
                MotionWrites++;
            if (address == FailWriteSymbol) throw new InvalidOperationException("模拟标量写入失败");
            if (address.EndsWith(".nAction")) authorityAction = (ushort)request.Value;
            if (address.EndsWith(".udiRequestSequence"))
            {
                AuthorityRequestWrites++;
                AuthorityAck = (uint)request.Value;
                if (authorityAction == 2 && ReleaseAllowed && !Session)
                    AuthorityOwner = Frame.ControlOwner = 0;
            }
            if (address.EndsWith("xWaveSession")) Session = (bool)request.Value;
            if (address == "GVL_motion.udiHeartbeat") WaveHeartbeatWrites++;
            if (address == "GVL_motion.nPointCount") { Frame.PointCount = (uint)request.Value; PointMetadataWrites++; }
            if (address == "GVL_motion.udiUploadSequence") sequence = (uint)request.Value;
            if (address == "GVL_motion.nUploadOffset") offset = (uint)request.Value;
            if (address == "GVL_motion.nUploadCount") size = (uint)request.Value;
            if (address == "GVL_motion.xClear" && (bool)request.Value)
            {
                ClearWrites++;
                Frame.ReceivedCount = Frame.LoadedCount = 0; Frame.DataReady = false; Frame.UploadStation = 1;
                Frame.Error = false; Frame.Stopped = Frame.Done = false; ValuesByAxis.Clear();
            }
            if (address.StartsWith("GVL_motion.aUpload["))
            {
                if (FailArrayWrite) throw new InvalidOperationException("模拟 ADS 写入失败");
                int axis = int.Parse(address.Split(',')[1]);
                if (!ValuesByAxis.ContainsKey(axis)) ValuesByAxis[axis] = new();
                ValuesByAxis[axis].Add((double)request.Value); ArrayWrites++;
                FrozenHeartbeat |= FreezeAfterArray;
            }
            if (address == "GVL_motion.xUpload" && (bool)request.Value && !HoldAcknowledgement)
            { Frame.ReceivedCount = offset + size; Frame.ReceivedSequence = sequence; }
            if (address == "GVL_motion.xAssemble" && (bool)request.Value)
            { Frame.LoadedCount = Frame.ReceivedCount; Frame.DataReady = true; }
            if (address == "GVL_motion.xStart" && (bool)request.Value)
            {
                StartWrites++; Frame.AcceptedSequence = sequence; Frame.TrajectoryBusy = true;
                Frame.Done = false; Frame.AxesIdle = false; runReads = 0;
                if (StartFaultCode != 0)
                {
                    Frame.Error = true; Frame.TrajectoryErrorId = StartFaultCode;
                    Frame.TrajectoryState = 90; Frame.TrajectoryBusy = false;
                }
            }
            if (address.Contains(".aAxis[") && address.EndsWith(".xStart") && (bool)request.Value)
                Frame.Axis1.Enabled = Frame.Axis2.Enabled = Frame.Axis3.Enabled = Frame.Axis4.Enabled = true;
            if (address.Contains(".aAxis[") && address.EndsWith(".gohome") && (bool)request.Value)
            {
                int axis = int.Parse(address.Split(".aAxis[")[1].Split(']')[0]);
                HomedAxes.Add(axis); Frame.AxesIdle = false;
                if (ChangeConfigOnHome) AxisLimits[3][0] = -201;
                if (!HoldHoming) { Frame.Axis2.Homed = true; Frame.AxesIdle = true; }
            }
            if (address == "GVL_motion.xStop" && (bool)request.Value)
            {
                Frame.TrajectoryBusy = false; Frame.UploadBusy = false; Frame.DataReady = false;
                Frame.Stopped = true; Frame.AxesStopped = true; Frame.AxesIdle = false;
            }
            if (address.Contains(".aAxis[") && address.EndsWith(".xStop") && !(bool)request.Value)
            { StopReleaseWrites++; Frame.AxesIdle = true; }
        }
        public PlcMachineMonitor Read()
        {
            if (!FrozenHeartbeat) Frame.Heartbeat++;
            if (Frame.TrajectoryBusy && ++runReads > 2)
            { Frame.TrajectoryBusy = false; Frame.Done = true; Frame.AxesIdle = true; }
            return Frame;
        }
    }
